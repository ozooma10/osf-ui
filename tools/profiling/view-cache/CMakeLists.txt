cmake_minimum_required(VERSION 3.25)
project(osfui_view_cache_bench LANGUAGES CXX)
set(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set(CACHE_SOURCE "${REPO}/src/Views/ViewCache.cpp" CACHE FILEPATH "Exact cache implementation to measure")
option(PROFILE_IDENTITY "Attribute filesystem identity checks in a generated source copy" OFF)
set(IMPLEMENTATION "${CACHE_SOURCE}")
if(PROFILE_IDENTITY)
    file(READ "${CACHE_SOURCE}" CONTENT)
    set(ANCHOR "std::filesystem::equivalent(it->path(), mod.cache.generation, ec)")
    string(FIND "${CONTENT}" "${ANCHOR}" FOUND)
    if(FOUND EQUAL -1)
        message(FATAL_ERROR "Cache identity instrumentation anchor changed")
    endif()
    string(REPLACE "${ANCHOR}" "CacheBenchEquivalent(it->path(), mod.cache.generation, ec)" CONTENT "${CONTENT}")
    set(IMPLEMENTATION "${CMAKE_CURRENT_BINARY_DIR}/ViewCache.profile.cpp")
    file(WRITE "${IMPLEMENTATION}" "#include <filesystem>\nbool CacheBenchEquivalent(const std::filesystem::path&, const std::filesystem::path&, std::error_code&);\n${CONTENT}")
endif()
add_executable(osfui_view_cache_bench main.cpp "${IMPLEMENTATION}")
target_compile_features(osfui_view_cache_bench PRIVATE cxx_std_23)
target_compile_options(osfui_view_cache_bench PRIVATE /EHsc /utf-8)
target_compile_definitions(osfui_view_cache_bench PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX CACHE_BENCH_PROFILE=$<BOOL:${PROFILE_IDENTITY}>)
target_include_directories(osfui_view_cache_bench PRIVATE "${REPO}/src")
